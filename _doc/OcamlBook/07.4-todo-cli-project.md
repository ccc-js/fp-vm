## 3. 核心程式碼

`lib/task.ml`：

```ocaml
type t = { id : int; title : string; done_ : bool }

let to_yojson { id; title; done_ } =
  `Assoc [ ("id", `Int id); ("title", `String title); ("done", `Bool done_) ]

let of_yojson = function
  | `Assoc f ->
      (match List.assoc_opt "id" f, List.assoc_opt "title" f, List.assoc_opt "done" f with
       | Some (`Int id), Some (`String title), Some (`Bool done_) -> Ok { id; title; done_ }
       | _ -> Error "bad task json")
  | _ -> Error "bad task json"

let show { id; title; done_ } =
  Printf.sprintf "[%s] #%d %s" (if done_ then "x" else " ") id title
```

`lib/store.ml`：

```ocaml
open Task

let path () =
  Filename.concat (Sys.getenv "HOME") ".todo.json"

let load () : (t list, string) result =
  let p = path () in
  if not (Sys.file_exists p) then Ok []
  else
    try
      let j = Yojson.Basic.from_file p in
      match j with
      | `List l ->
          List.fold_left
            (fun acc x ->
              match acc, Task.of_yojson x with
              | Ok ts, Ok t -> Ok (t :: ts)
              | Error e, _ -> Error e
              | _, Error e -> Error e)
            (Ok []) l
          |> Result.map List.rev
      | _ -> Error "root must be a list"
    with
    | Yojson.Json_error m -> Error ("json: " ^ m)
    | Sys_error m -> Error m

let save tasks =
  let j = `List (List.map Task.to_yojson tasks) in
  try Ok (Yojson.Basic.to_file (path ()) j)
  with Sys_error m -> Error m

let next_id tasks =
  List.fold_left (fun acc t -> max acc t.Task.id) 0 tasks + 1
```

`bin/main.ml`（Cmdliner 示意，完整版見 `Cmdliner.Term` 文件）：

```ocaml
open Cmdliner

let add title =
  match Store.load () with
  | Error e -> `Error (false, e)
  | Ok tasks ->
      let t = Task.{ id = Store.next_id tasks; title; done_ = false } in
      (match Store.save (tasks @ [ t ]) with
       | Ok () -> Printf.printf "added #%d\n" t.Task.id; `Ok ()
       | Error e -> `Error (false, e))

let add_cmd =
  let doc = "add a task" in
  let title = Arg.(required & pos 0 (some string) None & info [] ~docv:"TITLE") in
  Cmd.v (Cmd.info "add" ~doc) Term.(ret (const add $ title))

let list_cmd =
  let run () =
    match Store.load () with
    | Ok ts -> List.iter (fun t -> print_endline (Task.show t)) ts; `Ok ()
    | Error e -> `Error (false, e)
  in
  Cmd.v (Cmd.info "list") Term.(ret (const run $ const ()))

let () = exit (Cmd.eval (Cmd.group (Cmd.info "todo") [ add_cmd; list_cmd ]))
```

## 4. 測試與執行

```ocaml
(* test/test_task.ml *)
let test_roundtrip () =
  let t = Task.{ id = 1; title = "hi"; done_ = false } in
  match Task.of_yojson (Task.to_yojson t) with
  | Ok t2 -> Alcotest.(check string) "show" (Task.show t) (Task.show t2)
  | Error e -> Alcotest.fail e

let () = Alcotest.run "todo" [ ("task", [ Alcotest.test_case "roundtrip" `Quick test_roundtrip ]) ]
```
