# 6.3 first-class module 與泛型程式

functor 在編譯期決定模組；first-class module 讓模組成為執行期值。

## 1. 基本語法

```ocaml
module type SHOW = sig
  type t
  val show : t -> string
end

module IntShow = struct
  type t = int
  let show = string_of_int
end

module StrShow = struct
  type t = string
  let show s = "\"" ^ s ^ "\""
end

let print (type a) (module S : SHOW with type t = a) (v : a) =
  print_endline (S.show v)

let () =
  print (module IntShow) 42;
  print (module StrShow) "hi"
```

`(module S : SHOW ...)` 把模組打包為值，`(module IntShow)` 解包傳入。`type a.` 讓 `a` 與模組的 `t` 連動。

## 2. 執行期切換

```ocaml
let show_of_bool : (module SHOW with type t = int) =
  if Random.bool () then (module IntShow)
  else (module struct
    type t = int
    let show n = Printf.sprintf "0x%x" n
  end)

let () =
  let (module S) = show_of_bool in
  print_endline (S.show 255)
```

functor 做不到這種執行期分支。

## 3. 泛型工具：MonadUtils 範例

```ocaml
module type MONAD = sig
  type 'a t
  val return : 'a -> 'a t
  val bind : 'a t -> ('a -> 'b t) -> 'b t
end

module MonadUtils (M : MONAD) = struct
  let map f m = M.bind m (fun x -> M.return (f x))
  let rec map_m f = function
    | [] -> M.return []
    | x :: xs ->
        M.bind (f x) (fun y ->
          M.bind (map_m f xs) (fun ys -> M.return (y :: ys)))
end

module OptionMonad = struct
  type 'a t = 'a option
  let return x = Some x
  let bind m f = match m with None -> None | Some x -> f x
end

module OU = MonadUtils (OptionMonad)
let () = match OU.map_m (fun n -> if n mod 2 = 0 then Some (n/2) else None) [2;4] with
  | Some l -> Printf.printf "%s\n" (String.concat "," (List.map string_of_int l))
  | None -> print_endline "fail"
```

同一份 `MonadUtils` 可套 `Option`、`List`、`State`（functor 參數化）。

## 常見錯誤

-  quên `with type t = a` 導致抽象型別無法與外部連接。
- 在緊迴圈大量打包/解包模組：有效能成本，熱路徑改 functor。
